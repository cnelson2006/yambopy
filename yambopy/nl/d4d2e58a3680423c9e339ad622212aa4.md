print("""
STEP 2 -- CONVERT TO SI SHEET UNITS
Two problems with the raw chi^(2):
  (a) it is in Gaussian/esu units, while many SHG models use SI;
  (b) yambo averaged it over the WHOLE supercell, but most of the cell
      is vacuum - that dilutes the response by ~h/Lz.
chi2_supercell_to_sheet_SI() fixes both: esu->SI conversion, then
multiply by Lz to undo the vacuum dilution.  Result: the sheet
susceptibility chi_s^(2) in m^2/V - the standard 2D quantity.
""")

chi2_sheet = chi2_supercell_to_sheet_SI(chi2_g, Lz)   # m^2/V
chi2_bulk  = sheet_to_bulk_chi2(chi2_sheet, H)        # m/V (pm/V numbers)

i0 = np.argmin(np.abs(omega_eV - 0.7947))             # 1560 nm

plt.semilogy(omega_eV, np.abs(chi2_sheet), label='$|\\chi^{(2)}_s|$ (this run)')

print("""
STEP 4 -- THE SHG INTENSITY
Stack implements the structure-factor model (Song et al. 2023) for
air / 2D / film / substrate at normal incidence:

   I(2w) = 1/(2 eps0 c) |beta|^2 |2pi chi_s / lambda|^2 I0^2

where beta = (1+R_w)^2 (1+R_2w) encodes ALL the interference.
|beta|^2 ranges 0..64: near 64 the layers help you, near 0 they kill
the signal.  Here: the standard MoS2 / SiO2(285 nm) / Si wafer.
""")

stack = Stack(MoS2, silica, silicon, film_thickness=285e-9, h_2D=H)
beta  = stack.structure_factor(omega_eV)
I_shg = stack.shg_intensity(omega_eV, chi2_sheet, I0)

ok = ~np.isnan(beta)
print(f"  usable points  : {ok.sum()}/{len(omega_eV)}"
      "  (NaN = a layer lacks data at w or 2w there)")
print(f"  |beta|^2 range : {np.abs(beta[ok]).min()**2:.2f}"
      f" to {np.abs(beta[ok]).max()**2:.2f}  (0..64 possible)")

fig, ax = plt.subplots(1, 2, figsize=(12, 4))
ax[0].plot(omega_eV, np.abs(beta)**2, 'o-')
ax[0].set_title('interference factor $|\\beta|^2$')
ax[0].set_xlabel('$\\omega$ (eV)'); ax[0].grid(alpha=.3)
ax[1].semilogy(omega_eV, I_shg, 'o-')
ax[1].set_title('predicted SHG intensity')
ax[1].set_xlabel('$\\omega$ (eV)'); ax[1].set_ylabel('W/m$^2$')
ax[1].grid(alpha=.3, which='both')
plt.tight_layout(); plt.show()
